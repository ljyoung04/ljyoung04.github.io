```c
// Name: rot253.c
// Compile: gcc -fno-stack-protector rot253.c -o rot253 ; strip rot253
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>

#define MENU_ENCRYPTION 1
#define MENU_DECRYPTION 2

int rot253(uint8_t byte);
int rot253_inv(uint8_t byte);
void my_memcpy(char *dst, char *src, size_t n);
void hxd(uint8_t *mem, size_t n);
void unhexlify(char *output, char *hexstr, size_t len);
void encrypt(char *ct, char *pt);
void decrypt(char *pt, char *ct);
void print_menu();
void foo();

int main(void) {
    char yn[2];

    setvbuf(stdin, 0, _IONBF, 0);
    setvbuf(stdout, 0, _IONBF, 0);

    do {
        foo();

        printf("want to quit (y/n)? ");
        read(0, yn, 2);
    } while (yn[0] != 'y');

    return 0;
}

int rot253(uint8_t byte) {  // ROT-253
    char key;

    key = 253;
    return byte + key;
}

int rot253_inv(uint8_t byte) {  // inverse of ROT-253
    char key;

    key = 3;
    return byte + key;
}

void my_memcpy(char *dst, char *src, size_t n) {  // my custom memcpy *_*
    asm volatile(
        "push %rdi; push %rsi; push %rdx; push %rcx;"
        "mov %rdx, %rcx;"
        "rep movsb (%rsi),(%rdi);"
        "pop %rcx; pop %rdx; pop %rsi; pop %rdi"
    );
}

void hxd(uint8_t *mem, size_t n) {
    size_t i;

    for (i = 0; i < n; i++)
        printf("%02x", mem[i]);
    puts("\n");
}

void unhexlify(char *output, char *hexstr, size_t len) {
    int i;

    for (i = 0; i < len / 2; i++)
        sscanf(&hexstr[i * 2], "%2hhx", &output[i]);
}

void encrypt(char *ct, char *pt) {
    size_t len;
    int i;

    memset(pt, 0x00, 128);

    printf("plaintext? ");
    read(0, pt, 153);

    len = strlen(pt);

    my_memcpy(ct, pt, len);

    i = len;
    while (i--)
        ct[i] = rot253(ct[i]);

    printf("ciphertext (hexstr): ");
    hxd(ct, len);
}

void decrypt(char *pt, char *ct) {
    int i;
    char buf[255];
    ssize_t readn;

    memset(ct, 0x00, 128);
    memset(pt, 0x00, 128);

    printf("ciphertext (hexstr)? ");
    readn = read(0, buf, 254);
    if (readn < 1)
        exit(1);

    unhexlify(ct, buf, readn - 1);

    for (i = 0; i < 127; i++)
        pt[i] = rot253_inv(ct[i]);

    printf("plaintext: %s\n", pt);
}

void print_menu() {
    puts(" ===============");
    puts(" = ROT-253 *_* =");
    puts(" ===============");
    puts("1. encrypt");
    puts("2. decrypt");
    printf("> ");
}

void foo() {
    uint8_t menu;
    char pt[128];
    char ct[128];

    do {
        print_menu();
        scanf("%hhu", &menu);

        switch (menu) {
        case MENU_ENCRYPTION:
            encrypt(ct, pt);
            break;
        case MENU_DECRYPTION:
            decrypt(pt, ct);
            break;
        }
    } while (MENU_ENCRYPTION <= menu && menu <= MENU_DECRYPTION);

    memset(pt, 0x00, 144);
    memset(ct, 0x00, 128);
}

```

코드가 제공되기 때문에 분석해보자.

foo에서 128크기의 배열 2개를 선언한다.

그리고 encrypt함수를 살펴보면

pt에 153만큼 입력을 받고있다. pt의 크기는 128이기 때문에 버퍼 오버플로우가 발생한다.

그리고 pt의 길이를 strlen으로 구한다. 반환주소가 있는 부분까지 덮어버리면 반환주소 부분까지 길이를 측정할 것이기 때문에 hxd함수에서 릭이 가능하다. 포인터 형태로 전달하기 때문에 이 배열이 선언되어 있는 함수인 foo의 반환주소 부분이 릭될 것이라고 추측할 수 있다.

일단 시도해보니까 뭐가 출력되기는 하는데 이 부분도 암호화되어서 출력되는 것 같다. 하지만 하다 의문이 드는게 코드 중간을 보면

decrypted 함수에서도 입력을 길게 주면 rot253_inv 에서 버퍼의 크기를 벗어나서 출력해준다.

libc를 구하면 모든게 끝나는데 구할 방법이 안보인다..

일단 우리가 PIE 베이스를 구했으니까 가젯들과 코드 영역의 함수들을 계산해서 rao로 호출할 수 있다. 이걸 잘 활용해야할 것 같다.

페이로드의 길이는 153이 넘어가면 안된다. 더미로 136을 줘야하니까 사용할 수 있는 크기는 적다..  

ct에 /bin/sh 문자열 입력하고 바로 system 호출하면 될 것 같다. 진짜 libc 베이스만 구하면 끝이다.

decrypted 함수를 사용해야할 것 같다. 길이를 184만큼 줘서 메인 함수의 반환 함수 주소 까지 포함시켜서 복사 해오도록 하고 inv해서 보여주니까 그걸 다시 암호화하면 평문을 얻을 수 있다.

foo에서 sfp 전부를, ret을 1바이트 덮을 수 있기 때문에 sfp를 main의 버퍼로 해주고

ret을 1바이트 조작해서 main에서 read 할 때 rdx 설정을 건너뛰도록 하면 read 호출시 0x80만큼 입력을 받게되어 ROP가 가능하다. 여기서 가젯을 찾아서 hxd 함수로 라이브러리 베이스를 릭하고,

hxd 이후 레지스터 값들이 이상하기 때문에 다시 설정해주고, 다시 main의 read로 돌아가서 ROP를 하면 쉘을 딸 수 있다.
